def sort_idx(arr):   
    idx = [i for i in range(len(arr))] 
    for i in range(len(arr)-1):         
        for j in range(0, len(arr)-1-i): 
            a1 = arr[idx[j]][0]; a2 = arr[idx[j+1]][0] 
            t1 = arr[idx[j]][1]; t2 = arr[idx[j+1]][1] 
            if (a1 > a2) or (a1 == a2 and t1 > t2): 
                idx[j], idx[j+1] = idx[j+1], idx[j] 
    return idx
